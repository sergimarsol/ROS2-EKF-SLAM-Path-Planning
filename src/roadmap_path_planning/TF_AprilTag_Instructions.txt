HOW TO GET APRILTAG TF INFORMATION FOR OTHER PACKAGES
========================================================

This guide explains how to access AprilTag transform data from the TF system
in your own ROS2 packages, based on the hw2_solution implementation.

PREREQUISITES
=============
1. AprilTag detection system running (apriltag_ros package)
2. Camera system providing image feed
3. Static transform from robot base to camera
4. Known AprilTag positions in world coordinates (YAML config)

BASIC SETUP IN YOUR NODE
========================

1. IMPORTS
----------
```python
import rclpy
from rclpy.node import Node
from tf2_ros import Buffer, TransformListener
import numpy as np
import time
```

2. NODE INITIALIZATION
----------------------
```python
class YourNode(Node):
    def __init__(self):
        super().__init__('your_node_name')
        
        # TF setup
        self.tf_buffer = Buffer()
        self.tf_listener = TransformListener(self.tf_buffer, self)
        
        # Timer for checking transforms
        self.timer = self.create_timer(0.1, self.check_transforms)
```

3. BASIC TRANSFORM LOOKUP
-------------------------
```python
def check_transforms(self):
    """Check for AprilTag transforms"""
    try:
        # Replace 'tag_10' with your actual tag ID
        tag_frame = 'tag_10'
        
        # Get transform from base_link to tag
        transform = self.tf_buffer.lookup_transform(
            'base_link',           # target frame
            tag_frame,            # source frame  
            rclpy.time.Time()     # latest available time
        )
        
        # Extract position
        x = transform.transform.translation.x
        y = transform.transform.translation.y
        z = transform.transform.translation.z
        
        # Extract orientation (quaternion)
        qx = transform.transform.rotation.x
        qy = transform.transform.rotation.y
        qz = transform.transform.rotation.z
        qw = transform.transform.rotation.w
        
        self.get_logger().info(f"Tag at: ({x:.3f}, {y:.3f}, {z:.3f})")
        
    except Exception as e:
        # Transform not available
        pass
```

ADVANCED USAGE: MULTIPLE TAGS
==============================

1. LOAD TAG CONFIGURATION
--------------------------
```python
import yaml
import os
from ament_index_python.packages import get_package_share_directory

def load_tag_positions(self):
    """Load AprilTag world positions from YAML"""
    try:
        package_dir = get_package_share_directory('your_package_name')
        yaml_path = os.path.join(package_dir, 'config', 'apriltags.yaml')
        
        with open(yaml_path, 'r') as file:
            data = yaml.safe_load(file)
        
        self.tag_positions = {}
        for tag in data.get('apriltags', []):
            tag_id = tag.get('id')
            self.tag_positions[tag_id] = {
                'x': float(tag['x']),
                'y': float(tag['y']),
                'z': float(tag['z']),
                'qx': float(tag['qx']),
                'qy': float(tag['qy']),
                'qz': float(tag['qz']),
                'qw': float(tag['qw'])
            }
    except Exception as e:
        self.get_logger().error(f"Failed to load tags: {e}")
```

2. MULTI-TAG DETECTION WITH FILTERING
--------------------------------------
```python
def find_best_tag(self):
    """Find the closest/best visible AprilTag"""
    best_tag_id = None
    best_transform = None
    closest_distance = float('inf')
    
    for tag_id in self.tag_positions.keys():
        try:
            tag_frame = f'tag_{tag_id}'
            
            # Get transform
            observation = self.tf_buffer.lookup_transform(
                'base_link', 
                tag_frame, 
                rclpy.time.Time()
            )
            
            # Check transform freshness
            transform_time = rclpy.time.Time.from_msg(observation.header.stamp)
            time_diff = (self.get_clock().now() - transform_time).nanoseconds / 1e9
            
            if time_diff > 0.25:  # Skip old transforms (>250ms)
                continue
            
            # Calculate distance
            dx = observation.transform.translation.x
            dy = observation.transform.translation.y
            dz = observation.transform.translation.z
            distance = np.sqrt(dx*dx + dy*dy + dz*dz)
            
            # Keep closest tag
            if distance < closest_distance:
                closest_distance = distance
                best_tag_id = tag_id
                best_transform = observation
                
        except Exception:
            continue  # Tag not visible or transform unavailable
    
    return best_tag_id, best_transform
```

ROBOT LOCALIZATION FROM APRILTAG
=================================

1. COMPUTE ROBOT POSE FROM TAG
-------------------------------
```python
from scipy.spatial.transform import Rotation

def compute_robot_pose_from_tag(self, tag_id, tag_observation):
    """Compute robot position in world frame from tag observation"""
    
    # Get tag's known position in world
    tag_world = self.tag_positions[tag_id]
    tag_world_pos = np.array([tag_world['x'], tag_world['y'], tag_world['z']])
    tag_world_rot = Rotation.from_quat([
        tag_world['qx'], tag_world['qy'], 
        tag_world['qz'], tag_world['qw']
    ])
    
    # Get robot's observation of tag (tag relative to robot)
    obs_pos = np.array([
        tag_observation.transform.translation.x,
        tag_observation.transform.translation.y,
        tag_observation.transform.translation.z
    ])
    obs_rot = Rotation.from_quat([
        tag_observation.transform.rotation.x,
        tag_observation.transform.rotation.y,
        tag_observation.transform.rotation.z,
        tag_observation.transform.rotation.w
    ])
    
    # Inverse transformation: robot relative to tag
    tag_to_robot_rot = obs_rot.inv()
    tag_to_robot_pos = -tag_to_robot_rot.apply(obs_pos)
    
    # Transform to world frame
    robot_world_rot = tag_world_rot * tag_to_robot_rot
    robot_world_pos = tag_world_pos + tag_world_rot.apply(tag_to_robot_pos)
    
    # Extract yaw angle for 2D navigation
    yaw = robot_world_rot.as_euler('xyz')[2]
    
    return robot_world_pos[0], robot_world_pos[1], yaw
```

ERROR HANDLING AND BEST PRACTICES
==================================

1. ROBUST TRANSFORM LOOKUP
---------------------------
```python
def safe_transform_lookup(self, target_frame, source_frame, timeout=0.1):
    """Safely lookup transform with timeout"""
    try:
        # Use timeout for blocking lookup
        when = self.get_clock().now() - rclpy.duration.Duration(seconds=timeout)
        transform = self.tf_buffer.lookup_transform(
            target_frame,
            source_frame, 
            when,
            timeout=rclpy.duration.Duration(seconds=timeout)
        )
        return transform
    except Exception as e:
        self.get_logger().debug(f"Transform lookup failed: {e}")
        return None
```

2. TRANSFORM VALIDATION
-----------------------
```python
def is_transform_valid(self, transform, max_age=0.25, max_distance=5.0):
    """Validate transform quality"""
    
    # Check timestamp freshness
    transform_time = rclpy.time.Time.from_msg(transform.header.stamp)
    age = (self.get_clock().now() - transform_time).nanoseconds / 1e9
    
    if age > max_age:
        return False, "Transform too old"
    
    # Check reasonable distance
    pos = transform.transform.translation
    distance = np.sqrt(pos.x**2 + pos.y**2 + pos.z**2)
    
    if distance > max_distance:
        return False, "Tag too far away"
    
    return True, "Valid"
```

COMPLETE MINIMAL EXAMPLE
=========================

```python
#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from tf2_ros import Buffer, TransformListener
import numpy as np

class AprilTagReader(Node):
    def __init__(self):
        super().__init__('apriltag_reader')
        
        self.tf_buffer = Buffer()
        self.tf_listener = TransformListener(self.tf_buffer, self)
        
        # List of tag IDs to look for
        self.tag_ids = [10, 2, 3]
        
        self.timer = self.create_timer(0.5, self.check_tags)
        
    def check_tags(self):
        """Check for any visible AprilTags"""
        for tag_id in self.tag_ids:
            try:
                tag_frame = f'tag_{tag_id}'
                transform = self.tf_buffer.lookup_transform(
                    'base_link', 
                    tag_frame, 
                    rclpy.time.Time()
                )
                
                pos = transform.transform.translation
                self.get_logger().info(
                    f"Tag {tag_id}: ({pos.x:.2f}, {pos.y:.2f}, {pos.z:.2f})"
                )
                
            except Exception:
                pass  # Tag not visible

def main():
    rclpy.init()
    node = AprilTagReader()
    try:
        rclpy.spin(node)
    finally:
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
```

YAML CONFIGURATION FORMAT
==========================

Create apriltags.yaml in your package's config folder:

```yaml
apriltags:  
  - id: 10
    x: 1.4      # world position in meters
    y: 0.0
    z: 0.0
    qx: 0.5     # quaternion orientation
    qy: -0.5
    qz: -0.5
    qw: 0.5
  - id: 2
    x: 1.0
    y: 2.4
    z: 0.0
    qx: 0.707
    qy: 0.0
    qz: 0.0
    qw: 0.707
```

NOTES
=====

1. Frame names: AprilTag frames are named 'tag_X' where X is the tag ID
2. Transform direction: lookup_transform(target, source) gives source→target
3. Coordinate frames: Ensure your camera→base_link transform is correct
4. Time synchronization: Use rclpy.time.Time() for latest available transform
5. Error handling: Always wrap tf lookups in try/except blocks
6. Performance: Don't lookup transforms faster than detection rate (~30Hz)

For more complex scenarios, consider using tf2_geometry_msgs for coordinate 
transformations and tf2_sensor_msgs for point cloud transformations.
